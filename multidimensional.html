1. Enter the four number from user and print it into matrix

Solution

C Program

#include 

int main()
{
    int a[2][2];
    int i, j;

    printf("Enter four numbers:\n");

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            scanf("%d", &a[i][j]);
            j++;
        }
        i++;
    }

    printf("Matrix:\n");

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            printf("%d ", a[i][j]);
            j++;
        }
        printf("\n");
        i++;
    }

    return 0;
}

3 4

C Program

#include 

int main()
{
    int a[2][2], b[2][2], c[2][2];
    int i, j, k;

    printf("Enter first matrix:\n");

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            scanf("%d", &a[i][j]);
            j++;
        }
        i++;
    }

    printf("Enter second matrix:\n");

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            scanf("%d", &b[i][j]);
            j++;
        }
        i++;
    }

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            c[i][j] = 0;

            k = 0;
            while(k < 2)
            {
                c[i][j] = c[i][j] + a[i][k] * b[k][j];
                k++;
            }

            j++;
        }
        i++;
    }

    printf("Multiplication of matrix:\n");

    i = 0;
    while(i < 2)
    {
        j = 0;
        while(j < 2)
        {
            printf("%d ", c[i][j]);
            j++;
        }
        printf("\n");
        i++;
    }

    return 0;
}

3 4

Enter second matrix:

7 8
