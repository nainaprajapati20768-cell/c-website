Solution

C Program

#include <stdio.h>

int main()
{
    int a[2][2], i, j;

    printf("Enter 4 elements: ");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Array elements:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

}

4. Multiplication of Two Matrices

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[2][2], b[2][2], c[2][2] = {0};
    int i, j, k;

    printf("Enter first matrix: ");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix: ");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            for(k = 0; k < 2; k++)
            {
                c[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    printf("Multiplication:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }

    return 0;
}

{
    int a[2][2], i, j, sum = 0;

    printf("Enter 4 elements: ");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < 2; i++)

7. Find Sum of Each Row

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[2][2], i, j, sum;

    printf("Enter 4 elements: ");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        sum = 0;

        for(j = 0; j < 2; j++)
        {
            sum = sum + a[i][j];
        }

        printf("Row %d Sum = %d\n", i + 1, sum);
    }

    return 0;
}

Output

Enter 4 elements:
1 2
3 4
Row 1 Sum = 3
Row 2 Sum = 7
