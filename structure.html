1. Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

View Solution
Solution

C Program

#include 

struct employee
{
    int id;
    char name[30];
    char designation[30];
};

int main()
{
    struct employee e[50];
    int i = 0;

    while(i < 50)
    {
        printf("Enter employee ID: ");
        scanf("%d", &e[i].id);

        printf("Enter employee name: ");
        scanf("%s", e[i].name);

        printf("Enter designation: ");
        scanf("%s", e[i].designation);

        i++;
    }

    printf("\nID\tName\tDesignation\n");

    i = 0;
    while(i < 50)
    {
        printf("%d\t%s\t%s\n",
        e[i].id, e[i].name, e[i].designation);

        i++;
    }

    return 0;
}

Enter employee name: Rahul
Enter designation: Manager

Enter designation: Clerk

... Enter details of remaining employees ...

102     Aman    Clerk
... Remaining employee details ...

3. Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form ?

View Solution
Solution

C Program

#include 

struct employee
{
    int id;
    char name[30];
    char designation[30];
};

int main()
{
    struct employee e[50];
    int i = 0, j;

    while(i < 50)
    {
        printf("Enter employee ID: ");
        scanf("%d", &e[i].id);

        printf("Enter employee name: ");
        scanf("%s", e[i].name);

        printf("Enter designation: ");
        scanf("%s", e[i].designation);

        i++;
    }

    printf("\nID\tCharacter\n");

    i = 0;
    while(i < 50)
    {
        j = 0;

        while(e[i].name[j] != '\0')
        {
            printf("%d\t%c\n", e[i].id, e[i].name[j]);
            j++;
        }

        i++;
    }

    return 0;
}

Enter designation: Manager

4. Enter student details like roll name, fee, DOB and print all the detail by ascending order of roll number ?

View Solution
Solution

C Program

#include 

struct student
{
    int roll;
    char name[30];
    float fee;
    char dob[20];
};

int main()
{
    struct student s[50], temp;
    int n, i, j;

    printf("Enter number of students: ");
    scanf("%d", &n);

    i = 0;
    while(i < n)
    {
        printf("Enter roll number: ");
        scanf("%d", &s[i].roll);

        printf("Enter name: ");
        scanf("%s", s[i].name);

        printf("Enter fee: ");
        scanf("%f", &s[i].fee);

        printf("Enter DOB: ");
        scanf("%s", s[i].dob);

        i++;
    }

    i = 0;
    while(i < n)
    {
        j = i + 1;

        while(j < n)
        {
            if(s[i].roll > s[j].roll)
            {
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }

            j++;
        }

        i++;
    }

    printf("\nRoll\tName\tFee\tDOB\n");

    i = 0;
    while(i < n)
    {
        printf("%d\t%s\t%.2f\t%s\n",
        s[i].roll, s[i].name, s[i].fee, s[i].dob);

        i++;
    }

    return 0;
}
