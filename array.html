1. Input and Display 5 Array Elements

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Array elements: ");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output

Enter 5 numbers: 10 20 30 40 50
Array elements: 10 20 30 40 50

3. Find Largest Element in an Array

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i, max;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    max = a[0];

    for(i = 1; i < 5; i++)
    {
        if(a[i] > max)
            max = a[i];
    }

    printf("Largest = %d", max);

    return 0;
}

Output

Enter 5 numbers: 12 45 23 67 34
Largest = 67

5. Search an Element in an Array

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i, num, found = 0;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter number to search: ");
    scanf("%d", &num);

    for(i = 0; i < 5; i++)
    {
        if(a[i] == num)
        {
            found = 1;
            break;
        }
    }

    if(found == 1)
        printf("Element Found");
    else
        printf("Element Not Found");

    return 0;
}

Output

Enter 5 numbers: 10 20 30 40 50
Enter number to search: 30
Element Found

6. Count Even and Odd Elements in an Array

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i, even = 0, odd = 0;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);

        if(a[i] % 2 == 0)
            even++;
        else
            odd++;
    }

    printf("Even = %d\n", even);
    printf("Odd = %d", odd);

    return 0;
}

Output

Enter 5 numbers: 10 15 20 25 30
Even = 3
Odd = 2

7. Reverse Array Elements

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Reverse Array: ");

    for(i = 4; i >= 0; i--)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output

Enter 5 numbers: 10 20 30 40 50
Reverse Array: 50 40 30 20 10

8. Find Average of Array Elements

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i, sum = 0;
    float avg;

    printf("Enter 5 numbers: ");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
        sum = sum + a[i];
    }

    avg = sum / 5.0;

    printf("Average = %.2f", avg);

    return 0;
}

Output

Enter 5 numbers: 10 20 30 40 50
Average = 30.00
