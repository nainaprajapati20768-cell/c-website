Solution

{

    scanf("%d", &n);

    {
        printf("Enter value: ");

        printf("%d\n", b[i]);
    }

    return 0;
}

11. Enter five values of array and sort in ascending order?

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a[5], i, j, temp;

    for(i = 0; i < 5; i++)
    {
        printf("Enter value: ");
        scanf("%d", &a[i]);
    }

    for(i = 0; i < 5; i++)
    {
        for(j = i + 1; j < 5; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("Ascending order:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d\n", a[i]);
    }

    return 0;
}

Output

Enter value: 50
Enter value: 20
Enter value: 40
Enter value: 10
Enter value: 30
Ascending order:
10
20
30
40
50

Solution

{
    int a[100], n, i, max;

    printf("Enter size: ");

    for(i = 0; i < n; i++)
    {
        printf("Enter value: ");
