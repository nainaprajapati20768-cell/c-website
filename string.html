4. Enter a string from user and convert into title case ?

View Solution
Solution 
#include 

int main()
{
    char a[100];
    int i = 0;

    printf("Enter a string: ");
    scanf(" %[^\n]", a);

    if(a[0] >= 'a' && a[0] <= 'z')
    {
        a[0] = a[0] - 32;
    }

    while(a[i] != '\0')
    {
        if(a[i] == ' ')
        {
            if(a[i+1] >= 'a' && a[i+1] <= 'z')
            {
                a[i+1] = a[i+1] - 32;
            }
        }
        else if(a[i] >= 'A' && a[i] <= 'Z' && i != 0)
        {
            if(a[i-1] != ' ')
            {
                a[i] = a[i] + 32;
            }
        }

        i++;
    }

    printf("%s", a);

    return 0;
}

Output

Enter a string: hello world
Hello World

{

18. String palindrome example (using reverse) ?

View Solution
Solution 
#include 

int main()
{
    char a[100], b[100];
    int i = 0, j = 0;

    printf("Enter a string: ");
    scanf(" %[^\n]", a);

    while(a[i] != '\0')
    {
        i++;
    }

    i--;

    while(i >= 0)
    {
        b[j] = a[i];
        i--;
        j++;
    }

    b[j] = '\0';

    i = 0;

    while(a[i] == b[i] && a[i] != '\0')
    {
        i++;
    }

    if(a[i] == b[i])
    {
        printf("Palindrome");
    }
    else
    {
        printf("Not Palindrome");
    }

    return 0;
}

Output

Enter a string: madam
Palindrome
