6. Reverse a String

View Solution
Solution

C Program

#include <stdio.h>
#include <string.h>

int main()
{
    char str[50];
    int i;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Reverse String = ");

    for(i = strlen(str) - 1; i >= 0; i--)
    {
        printf("%c", str[i]);
    }

    return 0;
}

Output

Enter a string: Hello
Reverse String = olleH

9. Count Vowels in a String

Solution

C Program

#include <stdio.h>

int main()
{
    char str[50];
    int i, count = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == 'a' || str[i] == 'e' ||
           str[i] == 'i' || str[i] == 'o' ||
           str[i] == 'u' || str[i] == 'A' ||
           str[i] == 'E' || str[i] == 'I' ||
           str[i] == 'O' || str[i] == 'U')
        {
            count++;
        }
    }

    printf("Total Vowels = %d", count);

    return 0;
}

Output

Enter a string: Education
Total Vowels = 5

10. Check Palindrome String

View Solution
Solution

C Program

#include <stdio.h>
#include <string.h>

int main()
{
    char str[50];
    int i, len, flag = 0;

    printf("Enter a string: ");
    scanf("%s", str);

    len = strlen(str);

    for(i = 0; i < len / 2; i++)
    {
        if(str[i] != str[len - i - 1])
        {
            flag = 1;
            break;
        }
    }

    if(flag == 0)
        printf("Palindrome String");
    else
        printf("Not Palindrome String");

    return 0;
}

Output

Enter a string: madam
Palindrome String
