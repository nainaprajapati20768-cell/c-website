Solution

C Program

{

5. Arithmetic Operations

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    printf("Addition = %d\n", a + b);
    printf("Subtraction = %d\n", a - b);
    printf("Multiplication = %d\n", a * b);

    if (b != 0)
    {
        printf("Division = %.2f\n", (float)a / b);
        printf("Modulo = %d", a % b);
    }
    else
    {
        printf("Division by zero is not possible.");
    }

    return 0;
}

Output

Enter two numbers: 17 5
Addition = 22
Subtraction = 12
Multiplication = 85
Division = 3.40
Modulo = 2

7. Find Fractional Part of a Float Number

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    float num, fraction;

    printf("Enter a decimal number: ");
    scanf("%f", &num);

    fraction = num - (int)num;

    if (fraction < 0)
        fraction = -fraction;

    printf("Fractional part = %.2f", fraction);

    return 0;
}

Output

Enter a decimal number: 12.75
Fractional part = 0.75

9. Round Off a Float Number

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    float num;
    int result;

    printf("Enter a decimal number: ");
    scanf("%f", &num);

    if (num >= 0)
        result = (int)(num + 0.5);
    else
        result = (int)(num - 0.5);

    printf("Rounded number = %d", result);

    return 0;
}

Output

Enter a decimal number: 8.6
Rounded number = 9

10. Convert Lowercase into Uppercase

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char ch;

    printf("Enter a lowercase character: ");
    scanf("%c", &ch);

    if (ch >= 'a' && ch <= 'z')
    {
        ch = ch - 32;
    }

    printf("Uppercase = %c", ch);

    return 0;
}

Output

Enter a lowercase character: g
Uppercase = G

11. Calculate Total and Average of Five Subjects

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a, b, c, d, e, total;
    float average;

    printf("Enter marks of five subjects: ");
    scanf("%d %d %d %d %d", &a, &b, &c, &d, &e);

    total = a + b + c + d + e;
    average = total / 5.0;

    printf("Total = %d\n", total);
    printf("Average = %.2f", average);

    return 0;
}

Output

Enter marks of five subjects: 80 75 90 85 70
Total = 400
Average = 80.00

12. Calculate Simple Interest

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    float p, r, t, si;

    printf("Enter principal, rate and time: ");
    scanf("%f %f %f", &p, &r, &t);

    si = (p * r * t) / 100;

    printf("Simple Interest = %.2f", si);

    return 0;
}

Output

Enter principal, rate and time: 5000 5 2
Simple Interest = 500.00
