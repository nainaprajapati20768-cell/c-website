5. Write a program to enter 2 numbers from user add the number and show the answer, again take 2 number from user subtract the number display the answer, again take 2 number from user multiply the number display the answer and, again take 2 number from the user divide the number display the answer, again take 2 number find the remainder and display the answer ?

View Solution
Solution 
#include 

int main()
{
    int a, b;

    printf("Enter two numbers for addition: ");
    scanf("%d%d", &a, &b);
    printf("Addition = %d\n", a + b);

    printf("Enter two numbers for subtraction: ");
    scanf("%d%d", &a, &b);
    printf("Subtraction = %d\n", a - b);

    printf("Enter two numbers for multiplication: ");
    scanf("%d%d", &a, &b);
    printf("Multiplication = %d\n", a * b);

    printf("Enter two numbers for division: ");
    scanf("%d%d", &a, &b);
    printf("Division = %d\n", a / b);

    printf("Enter two numbers for remainder: ");
    scanf("%d%d", &a, &b);
    printf("Remainder = %d", a % b);

    return 0;
}

Output

Enter two numbers for addition: 10 5
Addition = 15
Enter two numbers for subtraction: 10 5
Subtraction = 5
Enter two numbers for multiplication: 10 5
Multiplication = 50
Enter two numbers for division: 10 5
Division = 2
Enter two numbers for remainder: 10 3
Remainder = 1

11. Write a program to enter marks of five subjects and calculate their average.

View Solution
Solution 
#include 

int main()
{
    int a, b, c, d, e;
    int sum;
    float average;

    printf("Enter marks of five subjects: ");
    scanf("%d%d%d%d%d", &a, &b, &c, &d, &e);

    sum = a + b + c + d + e;
    average = sum / 5.0;

    printf("Average = %.2f", average);

    return 0;
}

Output

Enter marks of five subjects: 80 70 90 60 100
Average = 80.00

12. Write a program to calculate simple interest using principal, rate and time entered by the user.

View Solution
Solution 
#include 

int main()
{
    float p, r, t, si;

    printf("Enter principal: ");
    scanf("%f", &p);

    printf("Enter rate: ");
    scanf("%f", &r);

    printf("Enter time: ");
    scanf("%f", &t);

    si = (p * r * t) / 100;

    printf("Simple Interest = %.2f", si);

    return 0;
}

Output

Enter principal: 1000
Enter rate: 5
Enter time: 2
Simple Interest = 100.00
