C Program

7. Find Grade Using Else-if Ladder

Solution

C Program

#include <stdio.h>

int main()
{
    int marks;

    printf("Enter marks: ");
    scanf("%d", &marks);

    if(marks >= 90 && marks <= 100)
        printf("Grade A");
    else if(marks >= 75 && marks < 90)
        printf("Grade B");
    else if(marks >= 60 && marks < 75)
        printf("Grade C");
    else if(marks >= 40 && marks < 60)
        printf("Grade D");
    else if(marks >= 0 && marks < 40)
        printf("Fail");
    else
        printf("Invalid Marks");

    return 0;
}

Enter marks: 85
Grade B

9. Simple Calculator Using Switch Statement

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int a, b;
    char op;

    printf("Enter operator (+, -, *, /): ");
    scanf(" %c", &op);

    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);

    switch(op)
    {
        case '+':
            printf("Addition = %d", a + b);
            break;

        case '-':
            printf("Subtraction = %d", a - b);
            break;

        case '*':
            printf("Multiplication = %d", a * b);
            break;

        case '/':
            if(b != 0)
                printf("Division = %.2f", (float)a / b);
            else
                printf("Cannot divide by zero");
            break;

        default:
            printf("Invalid Operator");
    }

    return 0;
}

Output

Enter operator (+, -, *, /): +
Enter two numbers: 10 20
Addition = 30

10. Check Vowel or Consonant Using Switch

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char ch;

    printf("Enter an alphabet: ");
    scanf(" %c", &ch);

    switch(ch)
    {
        case 'a':
        case 'e':
        case 'i':
        case 'o':
        case 'u':
        case 'A':
        case 'E':
        case 'I':
        case 'O':
        case 'U':
            printf("Vowel");
            break;

        default:
            if((ch >= 'a' && ch <= 'z') ||
               (ch >= 'A' && ch <= 'Z'))
                printf("Consonant");
            else
                printf("Not an Alphabet");
    }

    return 0;
}

Output

Enter an alphabet: A
Vowel
