23. Write a program to calculate area and perimeter of rectangle?

View Solution 
#include 
int main()
{
    int l,b,area,perimeter;

    printf("Enter length: ");
    scanf("%d",&l);

    printf("Enter breadth: ");
    scanf("%d",&b);

    area=l*b;
    perimeter=2*(l+b);

    printf("Area = %d\n",area);
    printf("Perimeter = %d",perimeter);
