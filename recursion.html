Recursion in C

Question 1: Write a program of recursion and display the output.

View Solution 
#include 

void show(int n)
{
    if(n > 0)
    {
        printf("%d ", n);
        show(n - 1);
    }
}

int main()
{
    show(5);

    return 0;
}
Output: 
5 4 3 2 1

Question 2: Write a program of recursion with global variable and display the output.

View Solution 
#include 

int n = 5;

void show()
{
    if(n > 0)
    {
        printf("%d ", n);
        n--;
        show();
    }
}

int main()
{
    show();

    return 0;
}
Output: 
5 4 3 2 1

Question 3: Write a program of recursion with local variable and display the output.

#include 

void show(int n)
{
    int a = n;

    if(a > 0)
    {
        printf("%d ", a);
        show(a - 1);
    }
}

int main()
{
    show(5);

    return 0;
}
Output:
