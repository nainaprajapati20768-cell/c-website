6. Write a program for printing your name on output screen but your name should blink 5 times.

View Solution
Solution

C Program

#include 

int main()
{
    int i = 1;

    while(i <= 5)
    {
        printf("\033[5mYourName\033[0m\n");
        i++;
    }

    return 0;
}

Output

YourName
YourName
YourName
YourName
YourName
Note: The blinking effect depends on terminal support. Some compilers may only show the name normally.

12. Enter password display like ****** ?

View Solution
Solution

C Program

#include 

int main()
{
    char password[50];
    int i = 0;

    printf("Enter password: ");
    scanf("%s", password);

    printf("Password: ");

    while(password[i] != '\0')
    {
        printf("*");
        i++;
    }

    return 0;
}

Output

Enter password: hello
Password: *****
Note: This program displays stars after the password is entered. It does not hide characters while typing.

    return 0;
}

Output

Enter two characters: A B
ASCII value of first character = 65
ASCII value of second character = 66

14. Print all ASCII character 0 to 127 ?

View Solution
Solution

C Program

#include 

int main()
{
    int i = 0;

    while(i <= 127)
    {
        if(i >= 32 && i <= 126)
        {
            printf("%d = %c\n", i, i);
        }
        else
        {
            printf("%d = Control Character\n", i);
        }

        i++;
    }

    return 0;
}

Output (sample)

0 = Control Character
1 = Control Character
2 = Control Character
...
32 =  
33 = !
34 = "
35 = #
...
65 = A
66 = B
67 = C
...
127 = Control Character
