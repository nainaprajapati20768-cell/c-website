2. Read a character from keyboard and display it.

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char ch;

    printf("Enter a character: ");
    scanf("%c", &ch);

    printf("Character = %c", ch);

    return 0;
}

Output

Enter a character: A
Character = A

3. Read four characters and display each on a separate line.

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char a, b, c, d;

    printf("Enter four characters: ");
    scanf(" %c %c %c %c", &a, &b, &c, &d);

    printf("%c\n", a);
    printf("%c\n", b);
    printf("%c\n", c);
    printf("%c\n", d);

    return 0;
}

Output

Enter four characters: A B C D
A
B
C
D

12. Display password using asterisks.

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char password[20];
    int i;

    printf("Enter password: ");
    scanf("%19s", password);

    printf("Password: ");

    for(i = 0; password[i] != '\0'; i++)
    {
        printf("*");
    }

    return 0;
}

Output

Enter password: abc123
Password: ******

13. Read two characters and print their ASCII values.

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    char a, b;

    printf("Enter two characters: ");
    scanf(" %c %c", &a, &b);

    printf("ASCII value of %c = %d\n", a, a);
    printf("ASCII value of %c = %d\n", b, b);

    return 0;
}

Output

Enter two characters: A B
ASCII value of A = 65
ASCII value of B = 66

14. Print ASCII characters from 0 to 127.

View Solution
Solution

C Program

#include <stdio.h>

int main()

    }

    return 0;
}
