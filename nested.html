7. Reverse a Number Using While Loop

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int n, rev = 0, rem;

    printf("Enter a number: ");
    scanf("%d", &n);

    while(n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    printf("Reverse = %d", rev);

    return 0;
}

Output

Enter a number: 1234
Reverse = 4321

8. Count Digits in a Number

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int n, count = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n == 0)
        count = 1;

    while(n != 0)
    {
        count++;
        n = n / 10;
    }

    printf("Total Digits = %d", count);

    return 0;
}

Output

Enter a number: 12345
Total Digits = 5

10. Check Whether a Number is Palindrome

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int n, temp, rev = 0, rem;

    printf("Enter a number: ");
    scanf("%d", &n);

    temp = n;

    while(n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    if(temp == rev)
        printf("Palindrome Number");
    else
        printf("Not Palindrome");

    return 0;
}

Output

Enter a number: 121
Palindrome Number

13. Check Prime Number Using Loop

View Solution
Solution

C Program

#include <stdio.h>

int main()
{
    int n, i, count = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        if(n % i == 0)
            count++;
    }

    if(n > 1 && count == 2)
        printf("Prime Number");
    else
        printf("Not Prime Number");

    return 0;
}

Output

Enter a number: 7
Prime Number
